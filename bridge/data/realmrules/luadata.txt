[mines]
; Gold seams. Freeciv's Gold never runs out; here each seam is worked out over seam_seasons seasons while a
; town of the land's owner lies close enough to work it, then the Gold is gone from the land and its yield
; with it (the bridge hears "seam_spent"). gold_house: the nation whose hills hold the realm's gold
; ("English" = House Goldcrest, "Wealth Is Strength"); gold_seams: how many seams are laid in its hills when
; a realm first meets this rule (other gold already on the map counts as seams too). on: "no" keeps
; Freeciv's own endless gold.
on = "yes"
seam_seasons = "40"
gold_house = "English"
gold_seams = "3"

[decrees]
; How many seasons a decree building stands (script.lua realm_decree). Their effects: effects.ruleset.
feast = "4"
relief = "2"
blight = "12"

[purse]
; The crown's purse each season (script.lua realm_purse). per_size: what a great house's towns yield the crown a
; season per size point at a tithe of 1 (at the usual 0.1, the realm's ~125 sizes bring ~95 gold a season).
; house_pays: the share of its own purse a house hands over (the rest is taken at its town gates). The Royal
; Court's upkeep each season: court_base, court_per_town for each crown town, court_share of the crown's purse.
per_size = "7.5"
house_pays = "0.25"
court_base = "2"
court_per_town = "2"
court_share = "0.02"

[raids]
; A raid's life (script.lua realm_raid): Freeciv's barbarians raised beside a town pillage the land around it each
; season, then leave after this many seasons.
seasons = "8"

[risings]
; A great house split by Freeciv's civil war (script.lua realm_rising, called by a story beat): the rebels' host
; (how many companies, of which kinds) and how many tiles it marches a season on the lord's seat. Taking the seat
; wins the rising (the rebels keep what they hold, a lesser house); else after `seasons` it settles itself (the side
; with more towns keeps them). Measured on the baseline (host 4, speed 1: three risings, each a split, the lord
; keeping 3, 4 and 3 towns; host 6, speed 2 marching on every town wiped a house out in 8 seasons).
host = "4"
host_kinds = "Pikemen,Archers,Knights"
speed = "1"
seasons = "8"
